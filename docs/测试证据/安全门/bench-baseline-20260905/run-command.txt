## 1) 官方镜像核验：Docker Hub 镜像内脚本 v1.3.4（过期）→ 上游 master v1.6.0

## 2) 源码构建（195:/root/p2-bench，任务后已删除）
cd /root/p2-bench && git clone --depth 1 https://github.com/docker/docker-bench-security.git
cd docker-bench-security && git rev-parse --short=9 HEAD              # 154869da6
docker build -q -t local/docker-bench-security:1.6.0-20260905 .       # sha256:9d64840ea608...

## 3) 跑分（一次性容器；首次尝试因 kernel 3.10 不支持 CAP_AUDIT_READ 失败一次，
##     去掉该 cap 后成功；audit 日志类检查(1.1.x)因此降级，如实记录）
cd /root/p2-bench && nohup docker run --rm --name p2-docker-bench \
  --label docker_bench_security \
  --net host --pid host --userns host \
  --cap-add AUDIT_CONTROL --cap-add SYS_PTRACE --cap-add SYS_ADMIN --cap-add DAC_READ_SEARCH \
  --memory=512m \
  -e DOCKER_CONTENT_TRUST=0 \
  -v /etc:/etc:ro \
  -v /usr/lib/systemd:/usr/lib/systemd:ro \
  -v /var/lib:/var/lib:ro \
  -v /var/run/docker.sock:/var/run/docker.sock:ro \
  -v /root/p2-bench/out:/out \
  local/docker-bench-security:1.6.0-20260905 \
  -l /out/docker-bench-security.log -p \
  > /root/p2-bench/out/stdout.txt 2>&1 &

# 运行窗口：2026-09-05 03:07:45 → 03:08:15 CST（约 30 秒）
# 产物：/out/docker-bench-security.log（原始文本，含 ANSI 色码）
#       /out/docker-bench-security.log.json（原始 JSON）
#       bench-log-clean.txt = 去色码净化版；stdout.txt = 控制台输出
# 容器退出后 --rm 自动清除（docker ps -a 复核无 p2-docker-bench 残留）

## 4) 清理（任务结束）
rm -rf /root/p2-bench
docker rmi local/docker-bench-security:1.6.0-20260905 docker/docker-bench-security:latest
